     1→/**
     2→ * Connec8 OS — CRUD registry.
     3→ *
     4→ * One typed registry powers /api/crud/[entity]/* for all business entities:
     5→ * field whitelisting + coercion, list filters, detail includes, and
     6→ * side-effect hooks (auto activity logging, lastActivityAt touch, etc.)
     7→ */
     8→import { db } from '@/lib/db'
     9→
    10→type FieldKind = 'string' | 'number' | 'boolean' | 'date' | 'json'
    11→
    12→export interface EntityConfig {
    13→  model: string
    14→  fields: Record<string, FieldKind>
    15→  required?: string[]
    16→  orderBy?: any
    17→  listInclude?: any
    18→  detailInclude?: any
    19→  listWhere?: (p: URLSearchParams) => any
    20→  postFilter?: (rows: any[], p: URLSearchParams) => any[]
    21→  listTransform?: (rows: any[], p: URLSearchParams) => any[] | Promise<any[]>
    22→  afterCreate?: (created: any, full: any) => Promise<void>
    23→  afterUpdate?: (before: any, after: any, full: any) => Promise<void>
    24→  beforeDelete?: (id: string) => Promise<void>
    25→}
    26→
    27→const now = () => new Date()
    28→const daysBetween = (a: Date, b: Date) => Math.floor((b.getTime() - a.getTime()) / 86_400_000)

    30→/** normalize channel / method → activity type */
    31→function outreachActivityType(channel: string): string {
    32→  switch (channel) {
    33→    case 'COLD_EMAIL': return 'EMAIL_SENT'
    34→    case 'WHATSAPP': return 'WHATSAPP_SENT'
    35→    case 'LINKEDIN': return 'LINKEDIN_SENT'
    36→    case 'PHONE': return 'CALL'
    37→    case 'VISIT': return 'VISIT_LOGGED'
    38→    default: return 'OUTREACH'
    39→  }
    40→}

    46→async function touchLead(leadId?: string | null, nextFollowUpAt?: Date | null) {
    47→  if (!leadId) return
    48→  await db.lead.update({ where: { id: leadId }, data: { lastActivityAt: now(), ...(nextFollowUpAt !== undefined ? { nextFollowUpAt } : {}) } }).catch(() => {})
    49→}
    50→async function touchSaasLead(saasLeadId?: string | null, nextFollowUpAt?: Date | null) {
    51→  if (!saasLeadId) return
    52→  await db.saasLead.update({ where: { id: saasLeadId }, data: { lastActivityAt: now(), ...(nextFollowUpAt !== undefined ? { nextFollowUpAt } : {}) } }).catch(() => {})
    53→}

    55→async function cascadeDeleteLead(id: string) {
    56→  await db.followUp.deleteMany({ where: { leadId: id } })
    57→  await db.outreach.deleteMany({ where: { leadId: id } })
    58→  await db.pitch.deleteMany({ where: { leadId: id } })
    59→  await db.meeting.updateMany({ where: { leadId: id }, data: { leadId: null } })
    60→  await db.contact.updateMany({ where: { leadId: id }, data: { leadId: null } })
    61→  await db.attachment.deleteMany({ where: { leadId: id } })
    62→  await db.activity.deleteMany({ where: { leadId: id } })
    63→}
    64→async function cascadeDeleteSaasLead(id: string) {
    65→  await db.followUp.deleteMany({ where: { saasLeadId: id } })
    66→  await db.outreach.deleteMany({ where: { saasLeadId: id } })
    67→  await db.pitch.deleteMany({ where: { saasLeadId: id } })
    68→  await db.meeting.updateMany({ where: { saasLeadId: id }, data: { saasLeadId: null } })
    69→  await db.contact.updateMany({ where: { saasLeadId: id }, data: { saasLeadId: null } })
    70→  await db.attachment.deleteMany({ where: { saasLeadId: id } })
    71→  await db.activity.deleteMany({ where: { saasLeadId: id } })
    72→}
    73→async function cascadeDeleteSaasLeadFull(id: string) { await cascadeDeleteSaasLead(id) }

   319→  products: {

   330→      const feats = (p.features || []) as any[]
   331→      const active = feats.filter((f) => f.status !== 'REJECTED')
   332→      const totalW = active.reduce((s, f) => s + (f.weight || 3), 0)
   333→      const doneW = active.filter((f) => f.status === 'COMPLETED').reduce((s, f) => s + (f.weight || 3), 0)
   334→      const byStatus: Record<string, number> = {}
   335→      feats.forEach((f) => { byStatus[f.status] = (byStatus[f.status] || 0) + 1 })
   336→      const leadActivity = (p.saasLeads || []).map((l: any) => new Date(l.lastActivityAt).getTime())
   337→      return {
   338→        id: p.id, name: p.name, tagline: p.tagline, description: p.description, status: p.status,
   339→        accent: p.accent, createdAt: p.createdAt,
   340→        stats: {
   341→          progress: totalW > 0 ? Math.round((doneW / totalW) * 100) : 0,
   342→          features: feats.length,
   343→          byStatus,
   344→          leads: p._count?.saasLeads ?? 0,
   345→          clients: p._count?.saasClients ?? 0,
   346→          lastActivityAt: leadActivity.length ? new Date(Math.max(...leadActivity)).toISOString() : p.updatedAt,
   347→        },
   348→      }
   349→    }),

   351→      features: { orderBy: [{ status: 'asc' }, { dateAdded: 'desc' }], include: { assignee: { select: { id: true, name: true } } } },

   728→/** Coerce + whitelist request body against the entity's field map. */
   729→export function coerceBody(cfg: EntityConfig, body: any, { partial }: { partial: boolean }): { data: any } | { error: string } {
   730→  if (!body || typeof body !== 'object') return { error: 'Invalid request body' }
   731→  const data: any = {}
   732→  for (const [field, kind] of Object.entries(cfg.fields)) {
   733→    if (!(field in body)) continue
   734→    let v = body[field]
   735→    if (kind === 'date') {
   736→      if (v === null || v === '') { data[field] = null; continue }
   737→      const d = new Date(v)
   738→      if (isNaN(d.getTime())) return { error: `Invalid date for field "${field}"` }
   739→      data[field] = d
   740→    } else if (kind === 'number') {
   741→      const n = Number(v)
   742→      if (Number.isNaN(n)) { data[field] = null } else { data[field] = n }
   743→    } else if (kind === 'boolean') {
   744→      data[field] = Boolean(v)
   745→    } else if (kind === 'json') {
   746→      if (v === null || v === '') { data[field] = null; continue }
   747→      data[field] = typeof v === 'string' ? v : JSON.stringify(v)
   748→    } else {
   749→      data[field] = v === '' ? null : String(v)
   750→    }
   751→  }
   752→  if (!partial) {
   753→    for (const req of cfg.required || []) {
   754→      if (data[req] === undefined || data[req] === null || data[req] === '') {
   755→        return { error: `Missing required field: ${req}` }
   756→      }
   757→    }
   758→  }
   759→  return { data }
   760→}
